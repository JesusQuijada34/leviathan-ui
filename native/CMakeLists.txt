cmake_minimum_required(VERSION 3.16)
project(LeviathanNative VERSION 1.0.5 LANGUAGES C)

set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)

add_library(levthcore SHARED src/liblevthcore.c)
target_include_directories(levthcore PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_compile_definitions(levthcore PRIVATE LEVTH_CORE_BUILD)
set_target_properties(levthcore PROPERTIES OUTPUT_NAME liblevthcore PREFIX "")

foreach(module setup window progressbar)
  add_library(levth${module} SHARED src/liblevth${module}.c)
  target_include_directories(levth${module} PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
  target_compile_definitions(levth${module} PRIVATE LEVTH_MODULE_BUILD)
  target_link_libraries(levth${module} PRIVATE levthcore)
  set_target_properties(levth${module} PROPERTIES OUTPUT_NAME liblevth${module} PREFIX "")
endforeach()

if(WIN32)
  target_link_libraries(levthwindow PRIVATE dwmapi user32)
  set(LEVTH_ICON_RC "${CMAKE_CURRENT_SOURCE_DIR}/resources/levth_version.rc")
  foreach(target levthcore levthsetup levthwindow levthprogressbar)
    target_sources(${target} PRIVATE ${LEVTH_ICON_RC})
  endforeach()
endif()

include(CTest)
if(BUILD_TESTING)
  add_executable(test_native tests/test_native.c src/liblevthcore.c
    src/liblevthsetup.c src/liblevthprogressbar.c src/liblevthwindow.c)
  target_include_directories(test_native PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/include)
  if(MSVC)
    target_compile_definitions(test_native PRIVATE LEVTH_CORE_BUILD LEVTH_MODULE_BUILD)
  endif()
  add_test(NAME native_abi COMMAND test_native)
endif()
